CREATE TABLE Employees (
    EmployeeName VARCHAR(50),
    Salary INT
);

INSERT INTO Employees (EmployeeName, Salary) VALUES
('Alice', 120000),
('Bob', 75000),
('Charlie', 45000);

SELECT 
    EmployeeName,
    Salary,
    CASE 
        WHEN Salary > 100000 THEN 'High Earner'
        WHEN Salary BETWEEN 50000 AND 100000 THEN 'Mid Earner'
        ELSE 'Low Earner'
    END AS SalaryCategory
FROM Employees;

CREATE TABLE Orders (
    OrderID INT,
    Status VARCHAR(50)
);

INSERT INTO Orders (OrderID, Status) VALUES
(1, 'Pending'),
(2, 'Shipped'),
(3, 'Pending'),
(4, 'Cancelled');

SELECT 
    CASE 
        WHEN Status = 'Pending' THEN 'Pending Orders'
        WHEN Status = 'Shipped' THEN 'Shipped Orders'
        ELSE 'Other Orders'
    END AS OrderStatus,
    COUNT(*) AS OrderCount
FROM Orders
GROUP BY 
    CASE 
        WHEN Status = 'Pending' THEN 'Pending Orders'
        WHEN Status = 'Shipped' THEN 'Shipped Orders'
        ELSE 'Other Orders'
    END;
