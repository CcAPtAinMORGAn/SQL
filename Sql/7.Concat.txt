
use yash



CREATE TABLE Employees (
    EmployeeID INT PRIMARY KEY,
    FirstName VARCHAR(50),
    LastName VARCHAR(50),
    HireDate DATETIME
);


INSERT INTO Employees (EmployeeID, FirstName, LastName, HireDate)
VALUES
(1, 'John', 'Doe', '2020-01-15'),
(2, 'Jane', 'Smith', '2021-06-10'),
(3, 'Sam', 'Johnson', '2022-03-25');


select * from Employees

SELECT EmployeeID, CONCAT(FirstName, ' ', LastName) AS FullName, HireDate
FROM Employees;


SELECT EmployeeID, CONCAT('First Name:- ',FirstName, '  ','Last Name:- ', LastName) AS FullName, HireDate
FROM Employees;


SELECT 
    EmployeeID,
    FirstName AS 'Name (First Name)', 
    LastName AS 'Name (Last Name)', 
    HireDate
FROM Employees;



SELECT EmployeeID, CONCAT('Hello ', FirstName, ' ', LastName) AS Greeting, HireDate
FROM Employees;

SELECT EmployeeID, CONCAT(FirstName, ' ', LastName, ' - Role:(Manager)') AS FullDetails, HireDate
FROM Employees;

SELECT EmployeeID, CONCAT('[', FirstName, ' ', LastName, ']') AS BracketedName, HireDate
FROM Employees;


SELECT *, 
       CONCAT(FirstName, ' ', LastName, ' was hired on ', FORMAT(HireDate, 'MMMM dd, yyyy')) AS EmployeeDetails
FROM Employees;

SELECT *, CONCAT('Yash Team: ', FirstName, ' ', LastName) AS "TeamList, HireDate"
FROM Employees;

SELECT EmployeeID, CONCAT(LOWER(FirstName), '.', LOWER(LastName), '@gmail.com') AS EmailAddress, HireDate
FROM Employees;
