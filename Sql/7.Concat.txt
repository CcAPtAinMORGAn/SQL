
use yash

CREATE TABLE Employees (
    EmployeeID INT PRIMARY KEY,
    FirstName VARCHAR(50),
    LastName VARCHAR(50),
    HireDate DATETIME
);


INSERT INTO Employees (EmployeeID, FirstName, LastName, HireDate)
VALUES

(2, 'Jane', 'Smith', '2021-06-10'),
(3, 'Sam', 'Johnson', '2022-03-25');


select * from Employees

SELECT EmployeeID, CONCAT(FirstName, ' ', LastName) AS FullName, HireDate
FROM Employees;


SELECT EmployeeID, CONCAT('First Name:- ',FirstName, '  ','Last Name:- ', LastName) AS FullName, HireDate
FROM Employees;

SELECT 
    EmployeeID,
    FirstName AS 'Full Name (First Name)', 
    LastName AS 'Full Name (Last Name)', 
    HireDate
FROM Employees;
