CREATE VIEW Total_Sales_and_Transactions AS
SELECT 
    a.Name AS Transaction_Name,
    a.Transactions as Transactions_count,
    
    b.Name AS Name_2,
    b.Total_Sales as Total_Sales  
FROM Transactions AS a
JOIN Total_Sales AS b
ON a.Name = b.Name;


select * from Total_Sales_and_Transactions

select Transaction_Name,Total_Sales,Transactions_count from Total_Sales_and_Transactions

CREATE VIEW Sorted_Total_Sales_and_Transactions AS select Transaction_Name,Total_Sales,Transactions_count from Total_Sales_and_Transactions
